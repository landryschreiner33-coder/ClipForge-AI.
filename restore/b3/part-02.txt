+# ------------------------------------------------------------------ the check
+def evaluate(ws: str, item: dict, settings: dict | None = None) -> dict:
+    """Check a preference on the reusable set (your edited clips of its kind, across formats), one clip left out at
+    a time: how far ClipForge's own choice was from your final edit (before) and how far the trend, read from the
+    other clips, would have been (after). {n, before, after, unit, by_format, passed, summary, at}."""
+    settings = settings if settings is not None else db.get_settings()
+    key = item["key"]
+    events = [e for e in _latest(ws, KEY_KIND[key]) if (e.get("data") or {}).get(key) is not None]
+    values = [_hashable(e["data"][key]) for e in events]
+    numeric = key in NUMERIC
+    rows = []
+    for i, e in enumerate(events):
+        actual = values[i]
+        base = _hashable(((e.get("data") or {}).get("base") or {}).get(key, _baseline(key, settings)))
+        others = values[:i] + values[i + 1:]
+        guess = trend(key, list(others), settings) if item["source"] == "inferred" else _hashable(item["value"])
+        guess = base if guess is None else guess  # no trend: ClipForge does what it did
+        if numeric:
+            a, b, g = _num(actual), _num(base), _num(guess)
+            if a is None:
+                continue
+            err_before, err_after = abs(a - (b or 0.0)), abs(a - (g if g is not None else b or 0.0))
+        else:
+            err_before, err_after = float(actual != _hashable(base)), float(actual != guess)
+        rows.append((e.get("fmt") or "highlights", err_before, err_after))
+    n = len(rows)
+    before = round(sum(r[1] for r in rows) / n, 3) if n else None
+    after = round(sum(r[2] for r in rows) / n, 3) if n else None
+    by_format = {}
+    for fmt in sorted({r[0] for r in rows}):
+        part = [r for r in rows if r[0] == fmt]
+        by_format[fmt] = {"n": len(part), "before": round(sum(r[1] for r in part) / len(part), 3),
+                          "after": round(sum(r[2] for r in part) / len(part), 3)}
+    passed = bool(n >= EVAL_MIN and before is not None and after is not None and before > 0
+                  and before - after >= PROMOTE * before)
+    unit = "s" if key in ("start_shift", "end_shift") else "" if numeric else "share"
+    if n == 0:
+        summary = "no clips to check yet"
+    elif numeric:
+        summary = f"off by {before:.2f}{unit} before, {after:.2f}{unit} with it"
+    else:
+        summary = (f"matched your choice in {round((1 - before) * 100)}% of clips before, "
+                   f"{round((1 - after) * 100)}% with it")
+    if n and n < EVAL_MIN:
+        summary += f"; needs {EVAL_MIN - n} more clip(s) to decide"
+    return {"n": n, "before": before, "after": after, "unit": unit, "by_format": by_format, "passed": passed,
+            "summary": summary, "at": round(time.time(), 3)}

+def taste_check(ws: str, folds: int = 5) -> dict:
+    """The clip-choice model (taste.py) checked on your own keep/leave-out choices: fitted on four fifths, tested on
+    the fifth left out, five times. Before: guessing the more common choice every time; after: the model's guess.
+    {n, before, after, passed, summary}: shares of choices it got right."""
+    import json as _json
+
+    from .. import taste
+
+    rows = [r for r in taste.events(ws) if r.get("features")]
+    n = len(rows)
+    if n < taste.MIN_EVENTS + 3:
+        return {"n": n, "before": None, "after": None, "passed": False,
+                "summary": f"needs {taste.MIN_EVENTS + 3 - n} more clip choice(s) to check"}
+    right_before = right_after = tested = 0
+    for k in range(folds):
+        test = rows[k::folds]
+        train_rows = [r for i, r in enumerate(rows) if i % folds != k]
+        m = taste.fit(train_rows, min_events=taste.MIN_EVENTS - 3, min_each=max(2, taste.MIN_EACH - 1))
+        if not m:
+            continue
+        common = sum(1 for r in train_rows if float(r["value"]) > 0) * 2 >= len(train_rows)
+        for r in test:
+            try:
+                x = _json.loads(r["features"])
+            except (TypeError, ValueError):
+                continue
+            kept = float(r["value"]) > 0
+            right_before += int(kept == common)
+            right_after += int((taste.predict(m, x)["p"] >= 0.5) == kept)
+            tested += 1
+    if not tested:
+        return {"n": n, "before": None, "after": None, "passed": False,
+                "summary": "not enough of both choices (kept and left out) to check yet"}
+    before, after = round(right_before / tested, 3), round(right_after / tested, 3)
+    return {"n": tested, "before": before, "after": after, "passed": after > before,
+            "summary": f"guessed {round(before * 100)}% of your choices right without it, {round(after * 100)}% with "
+                       f"it (each choice tested while left out)"}

+def shift_bounds(start: float, end: float, words: list[dict], opts: dict) -> tuple[float, float]:
+    """A new clip's start and end moved the way you usually move them (at most 3 s, onto a word's edge, never
+    shorter than the shortest clip allowed)."""
+    a, b = float(opts.get("start_shift") or 0), float(opts.get("end_shift") or 0)
+    if not a and not b:
+        return start, end
+    ns, ne = start + a, end + b
+    if a:
+        edges = [w["start"] for w in words if abs(w["start"] - ns) <= 0.6]
+        ns = min(edges, key=lambda t: abs(t - ns)) if edges else ns
+    if b:
+        edges = [w["end"] for w in words if abs(w["end"] - ne) <= 0.6]
+        ne = min(edges, key=lambda t: abs(t - ne)) if edges else ne
+    ns = max(0.0, ns)
+    if ne - ns < float(opts.get("min_duration") or 5) or ne <= ns:
+        return start, end
+    return round(ns, 3), round(ne, 3)

+                                                                              "value", "status", "evidence")}})

+    clip = db.get_clip(item.get("clip_id") or "") or {}
+    project = db.get_project(clip.get("project_id") or "") or {}
+    when = _local(item["planned_at"], settings).replace("T", " ") if item.get("planned_at") else ""
+    return " ".join(str(x or "") for x in (item.get("title"), clip.get("title"), project.get("name"),
+                                           PLATFORM_NAMES.get(item["platform"], item["platform"]), _account(item)[1],
+                                           when, _label(item.get("planned_at"), settings)))

+def _gap(settings: dict) -> float:
+    return 60.0 * float(settings.get("autopilot_min_gap_minutes", 0) or 0)
+
+
+def conflicts(settings: dict | None = None) -> dict[str, dict]:
+    """Posts not yet out that are closer than the minimum gap to another post on the same account: {post id: the
+    other post}. scheduler.reschedule refuses such a time; one can still arise when the gap or an account changes."""
+    settings = settings if settings is not None else _settings()
+    gap = _gap(settings)
+    if gap <= 0:
+        return {}
+    rows = [r for r in db.select("scheduled_publications", "status IN ('awaiting_approval', 'approved', "
+                                 "'publishing', 'reconciling', 'published') AND planned_at IS NOT NULL", (),
+                                 "planned_at ASC", 600)]
+    out: dict[str, dict] = {}
+    for i, a in enumerate(rows):
+        for b in rows[i + 1:]:
+            if b["planned_at"] - a["planned_at"] >= gap:
+                break
+            if a["platform"] != b["platform"] or _account(a)[0] != _account(b)[0]:
+                continue
+            for mine, other in ((b, a), (a, b)):  # the one that can still move is the one to fix
+                if mine["status"] in ("awaiting_approval", "approved") and mine["id"] not in out:
+                    out[mine["id"]] = other
+                    break
+    return out

+def _post_line(n: int, item: dict, settings: dict, clash: dict[str, dict]) -> str:
+    from ..autopilot import submissions
+
+    text = (f"{n}. {U+201C}{_post_name(item)}{U+201D} on {PLATFORM_NAMES.get(item['platform'], item['platform'])} "
+            f"({_account(item)[1]}): {_label(item.get('planned_at'), settings)}, "
+            f"{STATUS_WORDS.get(item['status'], item['status'])}")
+    if item.get("status_note"):
+        text += f" ({one_line(item['status_note'], 90)})"
+    if item["status"] in ("failed", "blocked", "action_needed") and (item.get("last_error") or item.get("fix")):
+        text += f". Why: {one_line(item.get('last_error') or '', 140)}"
+        if item.get("fix"):
+            text += f" What to do: {one_line(item['fix'], 140)}"
+    if item["id"] in clash:
+        other = clash[item["id"]]
+        text += (f". Too close to {U+201C}{_post_name(other)}{U+201D} at {_label(other['planned_at'], settings)} on the same "
+                 "account")
+    if item["status"] in ("approved", "awaiting_approval") and submissions.paused_for(item):
+        text += ". Its mission's uploads are paused"
+    return text + "."

+def find_post(args: dict, statuses: tuple[str, ...], doing: str) -> dict:
+    """The one post `args` names (post: words from its title, its video, its time like "18:30" or "Nov 26", or its
+    id; platform; account: words from the account's name), among posts in `statuses`. Raises ToolError naming the
+    candidates when none or several fit, so the model asks the owner which one."""
+    settings = _settings()
+    marks = ",".join("?" * len(statuses))
+    rows = db.select("scheduled_publications", f"status IN ({marks})", tuple(statuses),
+                     "planned_at IS NULL, planned_at ASC", 400)
+    want = str(args.get("post") or "").strip()
+    exact = [r for r in rows if r["id"] == want]
+    if exact:
+        return exact[0]
+    if args.get("platform"):
+        rows = [r for r in rows if r["platform"] == args["platform"]]
+    if args.get("account"):
+        who = set(_said_words(str(args["account"])))
+        rows = [r for r in rows if who & set(_said_words(_account(r)[1]))]
+    if not rows:
+        raise ToolError(f"There is no post that {doing}" + (" on that account or platform." if (
+            args.get("platform") or args.get("account")) else ". Use posts to see them all."))
+    if want:
+        words = {w for w in _said_words(want) if len(w) > 2} | set(re.findall(r"\d{1,2}:\d{2}", want))
+        scored = []
+        for r in rows:
+            have = _post_words(r, settings).lower()
+            hit = sum(1 for w in words if w in set(_said_words(have)) or (":" in w and w in have))
+            scored.append((hit, r))
+        best = max(h for h, _ in scored)
+        rows = [r for h, r in scored if h == best and h > 0]
+        if not rows:
+            raise ToolError(f"No post that {doing} matches {U+201C}{one_line(want, 60)}{U+201D}. They are: " + _names(
+                [r for _, r in scored], settings) + ".")
+    if len(rows) > 1:
+        raise ToolError(f"More than one post that {doing} fits: " + _names(rows, settings)
+                        + ". Ask the owner which one (its title, channel or time).")
+    return rows[0]

+                     f"{_label(r.get('planned_at'), settings)}" for r in rows[:6]) + ("; {U+2026}" if len(rows) > 6 else "")

+def _posts(ctx: Ctx, args: dict) -> Result:
+    settings = _settings()
+    view = args.get("view") or "upcoming"
+    statuses = VIEWS.get(view) or VIEWS["upcoming"]
+    marks = ",".join("?" * len(statuses))
+    order = "planned_at IS NULL, planned_at ASC" if view != "published" else "planned_at DESC"
+    rows = db.select("scheduled_publications", f"status IN ({marks})", statuses, order, 400)
+    if args.get("words"):
+        words = {w for w in _said_words(args["words"]) if len(w) > 2}
+        rows = [r for r in rows if words & set(_said_words(_post_words(r, settings)))] or rows
+    clash = conflicts(settings)
+    lines = [f"Times are in {_zone(settings)} (Settings {U+2192} Autopilot time zone); it is now "
+             f"{_label(time.time(), settings)} there. Posts on one account keep at least "
+             f"{_gap(settings) / 60:.0f} minutes apart."]
+    if not rows:
+        lines.append(f"No {view} posts.")
+    lines += [_post_line(i + 1, r, settings, clash) for i, r in enumerate(rows[:LIST_MAX])]
+    if len(rows) > LIST_MAX:
+        lines.append(f"{U+2026}and {len(rows) - LIST_MAX} more.")
+    if clash and view != "published":
+        lines.append(f"{len(clash)} post(s) are too close to another on the same account: fix_post_conflicts moves "
+                     "them to the nearest free times.")
+    return Result("\n".join(lines), f"Looked at your posts ({len(rows)})",
+                  {"links": [{"label": "Missions {U+2192} Schedule", "href": "#/missions"}]})
