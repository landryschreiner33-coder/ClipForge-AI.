+def _trim(clip: dict, start_shift: float, end_shift: float = 0.0) -> dict:
+    """The clip editor's save of a trim (what api.patch_clip hands to learned.observe)."""
+    from clipfoundry import db
+    from clipfoundry.brain import learned
+
+    before = db.get_clip(clip["id"])
+    edit = learned.mine(before.get("edit") or {}, {**(before.get("edit") or {}),
+                                                    "start": before["start"] + start_shift,
+                                                    "end": before["end"] + end_shift})
+    db.update_clip(clip["id"], edit=edit)
+    return {"events": learned.observe(before, db.get_clip(clip["id"])), "clip": db.get_clip(clip["id"])}
