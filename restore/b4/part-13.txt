+let sprite: HTMLCanvasElement | null = null;
+function shipSprite(): HTMLCanvasElement {
+  if (sprite) return sprite;
+  const c = document.createElement("canvas");
+  c.width = SHIP[0].length;
+  c.height = SHIP.length;
+  const g = c.getContext("2d");
+  if (g) {
+    SHIP.forEach((row, y) => [...row].forEach((ch, x) => {
+      if (ch === ".") return;
+      g.fillStyle = COLOURS[ch];
+      g.fillRect(x, y, 1, 1);
+    }));
+  }
+  return (sprite = c);
+}

+/** Dust in front of the ship: fixed points that the orbit carries past. */
+const DUST = Array.from({ length: 18 }, (_, i) => {
+  const r = (n: number) => ((Math.sin(i * 12.9898 + n * 78.233) * 43758.5453) % 1 + 1) % 1;
+  return { x: r(1), y: 0.15 + r(2) * 0.8, depth: 0.3 + r(3) * 0.7, size: r(4) };
+});
+
+/**
+ * One picture of the ship and dust on `g` (w {U+00D7} h canvas pixels) for the orbit pose, `t` seconds of flying (for the
+ * engine's flicker and the gentle bob; pass null for a still picture).
+ */
+export function paintShip(g: CanvasRenderingContext2D, w: number, h: number, pose: OrbitPose, t: number | null) {
+  g.clearRect(0, 0, w, h);
+  const s = Math.max(2, Math.round(h / 62));
+  const along = pose.phi / (Math.PI * 2);
+  // dust: drifts left as the ship flies right, the nearer specks faster and bigger
+  for (const d of DUST) {
+    const x = ((d.x - along * (1.5 + d.depth * 3)) % 1 + 1) % 1;
+    const size = Math.max(1, Math.round(s * (0.25 + d.depth * 0.45 + d.size * 0.2)));
+    g.fillStyle = `rgba(200, 225, 255, ${(0.18 + d.depth * 0.35).toFixed(2)})`;
+    g.fillRect(Math.round(x * w), Math.round(d.y * h), size, size);
+  }
+  const art = shipSprite();
+  const sw = art.width * s, sh = art.height * s;
+  const bob = t === null ? 0 : Math.sin(t * 0.9) * h * 0.012;
+  const cx = w * 0.19, cy = h * 0.74 + bob;
+  g.save();
+  g.translate(cx, cy);
+  // banks with the orbit, and dips its nose a little as the camera climbs
+  g.rotate(-pose.roll * 1.6 + (pose.incl - 0.13) * 0.3);
+  g.imageSmoothingEnabled = false;
+  // the engine: a flame of square pixels that flickers
+  const flick = t === null ? 0.6 : 0.55 + 0.45 * Math.abs(Math.sin(t * 13.0) * Math.sin(t * 7.3));
+  const flame = Math.round((3 + flick * 4) * s);
+  for (let k = 0; k < flame; k += s) {
+    const fade = 1 - k / flame;
+    const [r, gr] = [Math.round(150 + 105 * fade), Math.round(205 + 50 * fade)];
+    g.fillStyle = `rgba(${r}, ${gr}, 255, ${(0.85 * fade).toFixed(2)})`;
+    const half = Math.max(1, Math.round((1 + fade * 1.2) * s));
+    g.fillRect(-sw / 2 - s - k, -sh / 2 + 5 * s - half / 2 + s / 2, s, half);
+  }
+  g.drawImage(art, -sw / 2, -sh / 2, sw, sh);
+  // the side facing the hole is lit by the disk: brighter when its bright side is towards the ship
+  const warm = 0.18 + 0.14 * Math.cos(pose.phi);
+  g.globalCompositeOperation = "source-atop";
+  const grad = g.createLinearGradient(sw / 2, -sh / 2, -sw / 2, sh / 2);
+  grad.addColorStop(0, `rgba(255, 200, 120, ${warm.toFixed(2)})`);
+  grad.addColorStop(1, "rgba(255, 200, 120, 0)");
+  g.fillStyle = grad;
+  g.fillRect(-sw / 2, -sh / 2, sw, sh);
+  g.restore();
+}

+/**
+ * The robot office's Taser (the owner, 2026-11-25: "clicking it on a robot plays a brief cartoon spark and a funny
+ * reaction"). Purely for fun: nothing is sent anywhere and the robot's job carries on. The spark is drawn by
+ * OfficeMap (space.css .taser-zap); under Reduced motion it shows as a still spark. The sound plays only when
+ * Settings {U+2192} Appearance {U+2192} Sound effects is on.
+ */
+
+const REACTIONS = [
+  "Bzzt! My circuits!",
+  "Hey! I was rendering!",
+  "That tickles!",
+  "Ow, my bolts!",
+  "Totally fine. Totally fine.",
+  "Was that a firmware update?",
+  "My antenna is buzzing!",
+  "I'll put that in my report.",
+  "Static cling! Static cling!",
+  "Rebooting my dignity{U+2026}",
+];
+
+let last = -1;
+/** A reaction, never the same one twice in a row. */
+export function reaction(): string {
+  let k = Math.floor(Math.random() * REACTIONS.length);
+  if (k === last) k = (k + 1) % REACTIONS.length;
+  last = k;
+  return REACTIONS[k];
+}
+
+let audio: AudioContext | null = null;
+
+/** A short cartoon zap: crackling noise and a falling buzz, about a quarter of a second, quiet. */
+export function zapSound() {
+  try {
+    const Ctor = window.AudioContext || (window as unknown as { webkitAudioContext?: typeof AudioContext })
+      .webkitAudioContext;
+    if (!Ctor) return;
+    audio ||= new Ctor();
+    const ctx = audio, now = ctx.currentTime, length = 0.28;
+    const out = ctx.createGain();
+    out.gain.setValueAtTime(0.0001, now);
+    out.gain.exponentialRampToValueAtTime(0.16, now + 0.01);
+    out.gain.exponentialRampToValueAtTime(0.0001, now + length);
+    out.connect(ctx.destination);
+    // the crackle: noise through a band-pass filter
+    const noise = ctx.createBuffer(1, Math.ceil(ctx.sampleRate * length), ctx.sampleRate);
+    const data = noise.getChannelData(0);
+    for (let i = 0; i < data.length; i++) data[i] = (Math.random() * 2 - 1) * (Math.random() < 0.3 ? 1 : 0.2);
+    const crackle = ctx.createBufferSource();
+    crackle.buffer = noise;
+    const band = ctx.createBiquadFilter();
+    band.type = "bandpass";
+    band.frequency.value = 2400;
+    band.Q.value = 0.8;
+    crackle.connect(band).connect(out);
+    // the buzz: a sawtooth falling in pitch
+    const buzz = ctx.createOscillator();
+    buzz.type = "sawtooth";
+    buzz.frequency.setValueAtTime(880, now);
+    buzz.frequency.exponentialRampToValueAtTime(110, now + length);
+    const quiet = ctx.createGain();
+    quiet.gain.value = 0.35;
+    buzz.connect(quiet).connect(out);
+    crackle.start(now);
+    buzz.start(now);
+    crackle.stop(now + length);
+    buzz.stop(now + length);
+  } catch {
+    /* no sound: the spark still shows */
+  }
+}

+function useAudio() {
+  const audio = useRef<HTMLAudioElement | null>(null);
+  const url = useRef("");
+  const [playing, setPlaying] = useState("");
+  const stop = useCallback(() => {
+    audio.current?.pause();
+    audio.current = null;
+    if (url.current) URL.revokeObjectURL(url.current);
+    url.current = "";
+    setPlaying("");
+  }, []);
+  useEffect(() => stop, [stop]);
+  const play = async (key: string, v: { voice: string; speed: number; tone: number; text?: string }) => {
+    if (playing === key) {
+      stop();
+      return;
+    }
+    stop();
+    setPlaying(key);
+    try {
+      const { blob, note } = await textStories.hear(v);
+      if (note) toast(note);
+      url.current = URL.createObjectURL(blob);
+      const a = new Audio(url.current);
+      audio.current = a;
+      a.onended = () => audio.current === a && stop();
+      await a.play();
+    } catch (e) {
+      stop();
+      if ((e as Error).name !== "NotAllowedError") toast(errorText(e), true);
+    }
+  };
+  return { playing, play, stop };
