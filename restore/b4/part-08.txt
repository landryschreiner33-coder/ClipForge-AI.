+def clean_voice(raw: object, base: dict | None = None) -> dict:
+    """{"voice", "style", "speed", "tone"}: a style fills in speed and tone that were not given."""

+def clean_rules(raw: object) -> list[dict]:

+    m = _EDGE.match(token)

+def pronounce(text: str, rules: list[dict]) -> tuple[str, list[int]]:
+    """The words as the voice should read them, and for each written word how many spoken words stand for it (so
+    `words_back` can give the captions the written words). A rule matches whole words, any case, and may cover
+    several written words ("New York"); the longest rule wins. Punctuation around the words stays (the voice pauses
+    at it)."""
+    tokens = text.split()
+    ordered = sorted(({"words": r["word"].lower().split(), "say": r["say"].split()} for r in rules if r["say"].split()),
+                     key=lambda r: -len(r["words"]))
+    spoken: list[str] = []
+    counts: list[int] = []
+    i = 0
+    while i < len(tokens):
+        for r in ordered:
+            n = len(r["words"])
+            if i + n <= len(tokens) and [_core(t)[1].lower() for t in tokens[i:i + n]] == r["words"]:
+                parts = list(r["say"])
+                parts[0] = _core(tokens[i])[0] + parts[0]
+                parts[-1] = parts[-1] + _core(tokens[i + n - 1])[2]
+                spoken.extend(parts)
+                counts.extend([len(parts)] + [0] * (n - 1))
+                i += n
+                break
+        else:
+            spoken.append(tokens[i])
+            counts.append(1)
+            i += 1
+    return " ".join(spoken), counts

+def words_back(timed: list[dict], written: str, counts: list[int]) -> list[dict]:
+    """The written words with the times of the spoken words that stand for them ([{"w", "start", "end"}])."""
+    tokens = written.split()
+    out: list[dict] = []
+    k = 0
+    for token, n in zip(tokens, counts):
+        if n <= 0:
+            if out:
+                out[-1]["w"] = f"{out[-1]['w']} {token}"  # read together with the word before it
+            continue
+        part = timed[k:k + n]
+        k += n
+        if not part:
+            break
+        out.append({"w": token, "start": float(part[0]["start"]), "end": float(part[-1]["end"])})
+    return out

+def shape(src: Path, dst: Path, tone: float, cancel=None) -> None:
+    """The voice `tone` semitones higher (or lower), the same length: resampled, then stretched back (FFmpeg)."""
+    tone = min(TONE_RANGE[1], max(TONE_RANGE[0], float(tone)))

+                     cancel=cancel)
+    tmp.replace(dst)

+# Running a ClipForge worker (Full Cloud)
+
+A worker is ClipForge on a computer you rent (any Linux server or GPU machine you control). With **Settings {U+2192}
+Processing {U+2192} Full Cloud** on your desktop, Make clips runs there: the desktop uploads the video (or sends the link),
+the worker gets it, transcribes it, finds the moments, frames them, writes the captions and renders the clips, and
+the desktop brings the finished clips back into the project. Nothing is posted from the worker. Design and limits:
+[docs/clipforge/FULL_CLOUD.md](../docs/clipforge/FULL_CLOUD.md).
+
+Status: tested with a real worker process on the same machine (tests/test_full_cloud.py, the private test job).
+It has not been run on a rented server yet.
+
+## What the worker does and does not do
+
+- It answers **only** `/api/worker/v1/...` (with its key) and `/healthz`. Every other address, including the app's
+  pages and its whole API, is 404, so the worker exposes nothing else when it is on the internet.
+- It never posts, schedules, runs Autopilot, listens for phones or fetches the editing apps (these are off on a
+  worker without any setting).
+- A job's files stay on the worker until your desktop has brought the clips back (then they are deleted), or for
+  72 hours after the job ended, whichever comes first.
+- Your desktop sends the video, a transcript file when the project has one, and the clip choices from Settings
+  (length, count, captions, framing, sound). It never sends accounts, keys, posts or your Brain.
+
+## Set up (on the server)
+
+1. Install Python 3.12 and FFmpeg, copy the repository, then
+   `python3.12 -m venv .venv && .venv/bin/pip install -r requirements.txt`.
+   For fast transcription on an NVIDIA GPU, also `pip install -r requirements-gpu.txt`; on a CPU-only
+   server set the transcription device to CPU deliberately (below).
+2. Make a key (keep it like a password, never in chat or git): `openssl rand -base64 32`.
+3. Start the worker bound to this computer only, with the key in its environment:
+
+   ```sh
+   CLIPFOUNDRY_WORKER_KEY='<the key>' CLIPFOUNDRY_DATA=/var/lib/clipforge-worker \\
+   CLIPFOUNDRY_WORKERS=off .venv/bin/python -m clipfoundry --host 127.0.0.1 --port 8765
+   ```
